import pandas as pd
import numpy as np

def generate_mock_data():
    """Generates realistic customer churn dataset automatically"""
    np.random.seed(42)
    n = 1000
    d = {
        "customer_id": [f"CUST_{i:04d}" for i in range(n)],
        "age": np.random.randint(18, 70, size=n),
        "tenure": np.random.randint(1, 72, size=n),
        "monthly_charges": np.round(np.random.uniform(20.0, 120.0, size=n), 2),
        "contract_type": np.random.choice(["Month-to-month", "One year", "Two year"], size=n),
        "internet_service": np.random.choice(["DSL", "Fiber optic", "No"], size=n),
        "churn": np.random.choice([0, 1], size=n, p=[0.73, 0.27])
    }
    d["total_charges"] = np.round(d["tenure"] * d["monthly_charges"] + np.random.uniform(0, 50, size=n), 2)
    return pd.DataFrame(d)

def preprocess_data(df, is_training=True):
    """Cleans data and converts text categories to numbers"""
    df = df.copy()
    
    # Fill missing values if any
    df["total_charges"] = df["total_charges"].fillna(df["tenure"] * df["monthly_charges"])
    
    # Map text strings to exact numerical flags
    contract_map = {"Month-to-month": 0, "One year": 1, "Two year": 2}
    internet_map = {"No": 0, "DSL": 1, "Fiber optic": 2}
    
    df["contract_type"] = df["contract_type"].map(contract_map)
    df["internet_service"] = df["internet_service"].map(internet_map)
    
    features = ["age", "tenure", "monthly_charges", "total_charges", "contract_type", "internet_service"]
    
    if is_training:
        return df[features], df["churn"]
    return df[features]
